- Register the node's public QUIC address on-chain. Inside the pod the node sees
  only its pod IP, so peers learn the address only from its registry multiaddrs;
  without them, inbound goes through hole-punching and relays. Off-cluster, with
  the same keystore (`decdn setup` and `decdn node register` take the same
  --multiaddr flags; update-multiaddrs replaces the whole set):
{{- if eq $type "LoadBalancer" }}

     LB_IP=$(kubectl -n {{ $ns }} get svc {{ $name }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
{{- else if and (eq $type "NodePort") (not .Values.service.nodePort) }}

     NODE_PORT=$(kubectl -n {{ $ns }} get svc {{ $name }} -o jsonpath='{.spec.ports[0].nodePort}')
{{- end }}

     decdn node update-multiaddrs{{ range $addrs }} --multiaddr "{{ . }}"{{ end }}
